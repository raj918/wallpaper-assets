1. THE 3 JSON ENDPOINTS (jsDelivr Global CDN / Backend)
----------------------------------------------------------------------

ENDPOINT 1: VERSION CHECK
URL:
https://raw.githubusercontent.com/raj918/wallpaper-assets/main/images/data/version.json
(Or local backend: http://localhost:3000/version.json)

Method: GET
Response Format:
{
  "version": 14,
  "totalWallpapers": 1143,
  "categoriesCount": 14,
  "lastUpdated": "2026-10-05T03:53:04.145Z"
}

What it does:
Check this number when your mobile app opens. If the version is newer
than the stored local version in your app, fetch new data. Otherwise,
use cached data from local storage (instant loading, 0 data waste!).

3. READY-TO-USE CODE SNIPPET (REACT NATIVE / FLUTTER / JS)
----------------------------------------------------------------------

const CDN_BASE = "https://raw.githubusercontent.com/raj918/wallpaper-assets/main/images";

// 1. Fetch Version
export async function checkVersion() {
  try {
    const res = await fetch(`${CDN_BASE}/data/version.json?t=${Date.now()}`);
    const data = await res.json();
    return data.version;
  } catch (error) {
    console.error("Failed to fetch version:", error);
    return null;
  }
}

// 2. Fetch Categories (with designated covers & sequence order)
export async function fetchCategories() {
  try {
    const res = await fetch(`${CDN_BASE}/data/categories.json?t=${Date.now()}`);
    const data = await res.json();
    return data.categories; // Ordered array with name, index, count, coverThumbnail
  } catch (error) {
    console.error("Failed to fetch categories:", error);
    return [];
  }
}

// 3. Fetch Wallpapers
export async function fetchWallpapers() {
  try {
    const res = await fetch(`${CDN_BASE}/data/wallpapers.json?t=${Date.now()}`);
    const data = await res.json();
    return data;
  } catch (error) {
    console.error("Failed to fetch wallpapers:", error);
    return [];
  }
}

// 4. Helper: Get Designated Category Cover from Wallpapers list
export function getCategoryCoverUrl(categoryName, wallpapers) {
  const catItems = wallpapers.filter(w => w.category.toLowerCase() === categoryName.toLowerCase());
  const coverItem = catItems.find(w => w.isCover || (w.thumb && w.thumb.includes('cover=true')));
  return coverItem ? coverItem.thumb : (catItems[0] ? catItems[0].thumb : null);
}
